```python
async def _load_onchain_data(session: AsyncSession, asset_id: int) -> dict[str, float] | None:
    """Load latest on-chain data for a crypto asset."""
    from src.db.models import OnChainData
    today = date.today()
    result = await session.execute(
        select(OnChainData)
        .where(OnChainData.asset_id == asset_id)
        .where(OnChainData.date >= today - timedelta(days=1))
    )
    rows = result.scalars().all()
    if not rows:
        return None
    data: dict[str, float] = {}
    for row in rows:
        data[row.metric] = row.value
        data[f"{row.metric}_source"] = row.source  # type: ignore[assignment]
    return data

async def _load_github_data(session: AsyncSession, asset_id: int) -> dict[str, int | float] | None:
    """Load latest GitHub activity for a crypto asset."""
    from src.db.models import GitHubActivity
    today = date.today()
    result = await session.execute(
        select(GitHubActivity)
        .where(GitHubActivity.asset_id == asset_id)
        .where(GitHubActivity.date >= today - timedelta(days=1))
    )
    row = result.scalars().first()
    if not row:
        return None
    return {"stars": row.stars, "forks": row.forks, "weekly_commits": row.weekly_commits, "contributors": row.contributors}

async def _compute_correlation_data(session: AsyncSession, current_asset: Asset) -> dict[str, float] | None:
    """Compute rolling 30-day correlation between current asset and all watchlist assets."""
    from src.db.models import Watchlist
    # Get all watchlisted asset IDs
    result = await session.execute(select(Watchlist.asset_id).distinct())
    all_asset_ids = [r[0] for r in result.all()]
    if len(all_asset_ids) < 2:
        return None

    # Load 30-day close prices for all assets
    import numpy as np
    prices: dict[int, list[float]] = {}
    for aid in all_asset_ids:
        df = await _load_price_dataframe(session, type("A", (), {"id": aid})(), limit=30)  # type: ignore[arg-type]
        if not df.empty and len(df) >= 20:
            prices[aid] = df["close"].tolist()

    if len(prices) < 2 or current_asset.id not in prices:
        return None

    # Compute pairwise correlations with current asset
    current_prices = np.array(prices[current_asset.id])
    min_len = len(current_prices)
    correlations: list[float] = []
    for aid, p in prices.items():
        if aid == current_asset.id:
            continue
        other = np.array(p[:min_len])
        if len(other) < min_len:
            other = np.pad(other, (min_len - len(other), 0), constant_values=np.nan)
        valid = ~(np.isnan(current_prices) | np.isnan(other))
        if valid.sum() < 10:
            continue
        corr = np.corrcoef(current_prices[valid], other[valid])[0, 1]
        if not np.isnan(corr):
            correlations.append(float(corr))

    if not correlations:
        return None

    return {
        "avg_correlation": float(np.mean(correlations)),
        "max_correlation": float(np.max(np.abs(correlations))),
        "min_correlation": float(np.min(correlations)),
        "num_assets": len(correlations),
    }
```
