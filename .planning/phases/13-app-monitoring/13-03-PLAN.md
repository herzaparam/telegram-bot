   ```python
   @app.get("/health")
   async def health() -> dict:
       """Health check with DB connectivity, last pipeline run, and uptime."""
       db_ok = False
       try:
           async with async_session_factory() as session:
               await asyncio.wait_for(
                   session.execute(text("SELECT 1")),
                   timeout=5.0,
               )
               db_ok = True
       except Exception:
           pass

       last_run: str | None = None
       last_run_status: str | None = None
       try:
           async with async_session_factory() as session:
               result = await session.execute(
                   select(PipelineMetrics)
                   .order_by(PipelineMetrics.completed_at.desc())
                   .limit(1)
               )
               row = result.scalar_one_or_none()
               if row:
                   last_run = row.completed_at.isoformat()
                   last_run_status = row.status
       except Exception:
           pass

       uptime_seconds = round(time.monotonic() - _boot_time, 1)

       return {
           "status": "ok" if db_ok else "degraded",
           "db": "connected" if db_ok else "disconnected",
           "last_pipeline_run": last_run,
           "last_pipeline_status": last_run_status,
           "uptime_seconds": uptime_seconds,
       }
   ```

4. **Add missed-run checker background task** (per D-06):
   ```python
   async def _missed_run_checker() -> None:
       """Periodic check for missed pipeline runs. Runs every 15 minutes."""
       await asyncio.sleep(60)  # Initial delay to let system settle
       while True:
           try:
               now = datetime.now(timezone.utc)
               expected_hour = settings.expected_pipeline_hour
               grace_minutes = settings.missed_run_grace_minutes

               # Only check if we're past the expected window
               expected_minute = expected_hour * 60 + grace_minutes
               current_minute = now.hour * 60 + now.minute
               if current_minute >= expected_minute:
                   async with async_session_factory() as session:
                       result = await session.execute(
                           select(PipelineMetrics)
                           .where(PipelineMetrics.run_date == now.date())
                           .limit(1)
                       )
                       todays_run = result.scalar_one_or_none()
                       if todays_run is None:
                           await send_error_alert(
                               "bot",
                               "MissedPipelineRun",
                               f"No pipeline run completed for {now.date()}. "
                               f"Expected by {expected_hour:02d}:{grace_minutes:02d} UTC.",
                           )
           except Exception:
               logger.exception("missed_run_check_error")
           await asyncio.sleep(900)  # Check every 15 minutes
   ```

1. **Create src/bot/handlers/stats.py** (per D-09):
   ```python
   """Handler for /stats command — pipeline performance trends."""

   from datetime import date, timedelta

   import structlog
   from sqlalchemy import select
   from telegram import Update
   from telegram.ext import ContextTypes

   from src.bot.auth import is_authorized
   from src.db.database import async_session_factory
   from src.db.models import PipelineMetrics

   logger = structlog.get_logger(__name__)


   async def stats_handler(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
       """Show recent pipeline performance trends."""
       if not is_authorized(update):
           return

       try:
           async with async_session_factory() as session:
               cutoff = date.today() - timedelta(days=7)
               result = await session.execute(
                   select(PipelineMetrics)
                   .where(PipelineMetrics.run_date >= cutoff)
                   .order_by(PipelineMetrics.run_date.desc())
                   .limit(7)
               )
               rows = result.scalars().all()

           if not rows:
               await update.message.reply_text(
                   "No pipeline metrics available yet.",
                   parse_mode="HTML",
               )
               return

           lines = ["<b>Pipeline Stats (Last 7 Days)</b>\n"]

           for row in rows:
               status_icon = {
                   "completed": "\u2705",
                   "partial": "\u26a0\ufe0f",
                   "failed": "\u274c",
               }.get(row.status, "\u2753")

               duration_min = row.total_duration_seconds / 60
               tokens = row.llm_prompt_tokens + row.llm_completion_tokens
               lines.append(
                   f"{status_icon} <b>{row.run_date}</b> "
                   f"| {duration_min:.1f}m "
                   f"| {row.assets_processed} assets "
                   f"({row.assets_failed} failed) "
                   f"| {tokens:,} tokens"
               )
               if row.alert_count > 0:
                   lines.append(f"   \u26a0\ufe0f {row.alert_count} alerts")

           # Summary stats
           avg_duration = sum(r.total_duration_seconds for r in rows) / len(rows) / 60
           total_tokens = sum(r.llm_prompt_tokens + r.llm_completion_tokens for r in rows)
           success_count = sum(1 for r in rows if r.status == "completed")

           lines.append(f"\n<b>Summary:</b>")
           lines.append(f"Success rate: {success_count}/{len(rows)} runs")
           lines.append(f"Avg duration: {avg_duration:.1f} min")
           lines.append(f"Total tokens (7d): {total_tokens:,}")

           await update.message.reply_text(
               "\n".join(lines),
               parse_mode="HTML",
           )
       except Exception:
           logger.exception("stats_handler_error")
           await update.message.reply_text(
               "Error fetching pipeline stats.",
               parse_mode="HTML",
           )
   ```
