```python
@dataclass(frozen=True)
class Signal:
    category: str
    score: float           # -1.0 to +1.0
    confidence: float      # 0.0 to 1.0
    reasoning: str
    indicators: dict
    data_quality: dict

class BaseEngine(ABC):
    @abstractmethod
    def analyze(self, asset_id: int, asset_symbol: str, df: pd.DataFrame) -> Signal: ...
    @property
    @abstractmethod
    def category(self) -> str: ...
    @property
    def supports_stocks(self) -> bool: return True
    @property
    def supports_crypto(self) -> bool: return True
```

       ```python
    import math
    import pandas as pd
       ```

    ```python
    from src.engines.technical import TechnicalEngine, _rsi_to_score, _bollinger_to_score, compute_technical_indicators

        def setup_method(self):
            self.engine = TechnicalEngine()

        def test_category_is_technical(self):
            assert self.engine.category == "technical"

        def test_supports_stocks(self):
            assert self.engine.supports_stocks is True

            assert self.engine.supports_crypto is True

    class TestAnalyzeWith200Rows:
        def test_returns_valid_signal(self, sample_price_df_200):
            engine = TechnicalEngine()
            signal = engine.analyze(1, "BBCA", sample_price_df_200)
            assert -1.0 <= signal.score <= 1.0
            assert 0.0 <= signal.confidence <= 1.0
            assert signal.category == "technical"
            assert len(signal.reasoning) > 0

        def test_indicators_dict_has_expected_keys(self, sample_price_df_200):
            engine = TechnicalEngine()
            signal = engine.analyze(1, "BBCA", sample_price_df_200)
            expected_keys = {"rsi_14", "rsi_7", "macd_line", "ema_9", "ema_200", "obv", "volume_ratio"}
            assert expected_keys.issubset(set(signal.indicators.keys()))

        def test_data_quality_has_trading_days(self, sample_price_df_200):
            engine = TechnicalEngine()
            signal = engine.analyze(1, "BBCA", sample_price_df_200)
            assert "trading_days" in signal.data_quality
            assert signal.data_quality["trading_days"] == 200

    class TestInsufficientData:
        def test_10_rows_lower_confidence(self, sample_price_df_10):

            signal = engine.analyze(1, "BBCA", sample_price_df_10)

            assert len(signal.data_quality["indicators_skipped"]) > 0

        def test_empty_df_returns_zero(self, empty_price_df):
            engine = TechnicalEngine()
            signal = engine.analyze(1, "BBCA", empty_price_df)
            assert signal.score == 0.0
            assert signal.confidence == 0.0

    class TestScoreRange:
        def test_score_always_clamped(self, sample_price_df_200):
            engine = TechnicalEngine()
            signal = engine.analyze(1, "BBCA", sample_price_df_200)
    ```
